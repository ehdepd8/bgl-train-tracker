function renderRideLogs() { const container = document.getElementById('train-ride-logs'); if (!container) return; // 1. HTML을 새로 그리기 전, 현재 사용자의 스크롤 위치를 미리 기억 const prevScrollTop = container.scrollTop; const currentRideLogs = teamsData[currentTeamKey].rideLogs || []; if (currentRideLogs.length === 0) { container.innerHTML = `기록 없음`; return; } container.innerHTML = currentRideLogs.map(l => { const isRiding = !l.endTime; const durationText = isRiding ? getElapsedDurationText(l.startTimestamp, l.startTime) : ''; return `
${l.trainNum}편성 ${l.staffName}
${isRiding && durationText ? ` ${durationText} ` : ''}
${l.startTime} ~ ${isRiding ? '탑승중' : l.endTime}
`; }).join(''); // 2. 맨 위에 있었으면(10px 이내) 정확히 0(맨 꼭대기)으로 고정, // 내려서 보고 있었으면 보던 위치 그대로 완벽 복원 if (prevScrollTop <= 10) { container.scrollTop = 0; } else { container.scrollTop = prevScrollTop; } }
